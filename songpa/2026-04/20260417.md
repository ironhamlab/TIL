## 2. 컴포넌트(Component)
### 2-5. Iteration
#### 2-5-1. 렌더링 리스트

</br>

* `arr.map(callback);` : callback함수의 파라미터로 전달 된 배열 내 각 요소를 **원하는 규칙에 따라 변환 후 그 결과로 새로운 배열을 반환**한다.

</br>

* **callback 함수**: 새로운 배열 요소를 생성하는 규칙을 담은 함수로 파라미터는 3가지를 가질 수 있다.
    - currentValue: 현재 처리하고 있는 요소
    - index: 현재 처리하고 있는 요소의 index 값
    - array: 현재 처리하고 있는 원본 배열



</br>

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Document</title>
    <script crossorigin src="https://unpkg.com/react@18/umd/react.development.js"></script>
    <script crossorigin src="https://unpkg.com/react-dom@18/umd/react-dom.development.js"></script>
    <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>
</head>
<body>
    <div id="root"></div>

    <script type="text/babel">

        const fruits = ['사과', '바나나', '포도', '오렌지'];

        function FruitList() {
            return (
                <div>
                    <h2>과일 목록</h2>
                    <ul>
                        {fruits.map( (fruit, index) => (
                            <li key={index}>{fruit}</li>
                        ) )}
                    </ul>
                </div>
            )    
        }

        const users = [
            { id: 1, name:  '홍길동' },
            { id: 2, name: '유관순'},
            { id: 3, name: '이순신'}
        ]

        function UserList() {
            return (
                <>
                    <h2>회원 목록</h2>
                    <ul>
                        {users.map(user => (
                            <li key={user.id}>
                                {user.name}
                            </li>
                        ))}
                    </ul>
                </>
            )
        }

        function App() {
            return (
                <>
                    <FruitList />
                    <UserList />
                </>
            )
        }


        ReactDOM.createRoot(document.getElementById('root')).render(<App />);
    
    </script>
    
</body>
</html>
```

#### 2-5-2. 리스트 추가/삭제

</br>

* `filter`:기존 배열에서 callback 함수를 각 요소마다 실행하고, **true를 반환하는 요소들로만 새로운 배열을 반환** 한다.

</br>

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Document</title>
    <script crossorigin src="https://unpkg.com/react@18/umd/react.development.js"></script>
    <script crossorigin src="https://unpkg.com/react-dom@18/umd/react-dom.development.js"></script>
    <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>
</head>
<body>
    <div id="root"></div>

    <script type="text/babel">

        const { useState } = React;

        function NameList() {

            const [inputValue, setInputValue] = useState('');
            const [names, setNames] = useState([{id: 1, text: '판다'}]);
            const [nextId, setNextId] = useState(2);

            const handleChange = (e) => setInputValue(e.target.value);
            
            const handleAdd = () => {
                if (inputValue.trim() === '') return;
                const newName = {id: nextId, text: inputValue};

                setNames([...names, newName]);
                setNextId(nextId + 1);
                setInputValue('');
            };

            const handleRemove = (idToRemove) => {
                /*
                filter:기존 배열에서 callback 함수를 각 요소마다 실행하고,
                        true를 반환하는 요소들로만 새로운 배열을 반환
                */
               // id로 넘어온 값과 일치하지 않는 객체만 다시 배열로 반환
               const updateNames = names.filter(name => name.id !== idToRemove);
               setNames(updateNames);
            }

            return(
                <>
                    <input type="text" value={inputValue} onChange={handleChange} placeholder="이름을 입력하세요" />
                    <button onClick={handleAdd}>추가</button>

                    <ul>
                        {names.map(name => (
                            <li key={name.id}>
                                {name.text}
                                <button onClick={() => handleRemove(name.id)}>삭제</button>
                            </li>
                        ))}
                    </ul>
                </>
            )
        }


        ReactDOM.createRoot(document.getElementById('root')).render(<NameList />);
    
    </script>
    
</body>
```

#### 2-5-3. Todo 리스트

</br>

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Document</title>
    <script crossorigin src="https://unpkg.com/react@18/umd/react.development.js"></script>
    <script crossorigin src="https://unpkg.com/react-dom@18/umd/react-dom.development.js"></script>
    <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>
</head>
<body>
    <div id="root"></div>

    <script type="text/babel">

        const { useState } = React;

        // TodoItem 컴포넌트(UI)
        // props: todo, onRemove, onToggle
        function TodoItem({todo, onRemove, onToggle}) {
            return (
                <p>
                    <input type="checkbox" checked={todo.done} onChange={onToggle} />
                    <span style={{textDecoration: todo.done ? 'line-through' : 'none'}}>{todo.text}</span>
                    <button onClick={onRemove}>x</button>
                </p>
            )
        }
        
        // TodoList 컴포넌트(상태+로직)
        function TodoList() {
            // 상태(state) 만들기
            const [inputValue, setInputValue] = useState('');
            const [todos, setTodos] = useState([]);
            const [id, setId] = useState(1);

            // 함수들 구현

            // 입력값 변경
            const handleChange = (e) => setInputValue(e.target.value);
            
            // 추가
            const handleAdd = () => {
                if (inputValue.trim()==='') return;
                const newTodo = {id: id, text: inputValue, done: false};

                setTodos([...todos, newTodo]);
                setId(id + 1);
                setInputValue('');
            };

            const handleRemove = (id) => {
                setTodos(todos.filter(todo => todo.id !== id));
            }
            const handleToggle = (id) => {
                setTodos(
                    todos.map(todo =>
                        todo.id === id
                        ? {...todo, done: !todo.done }
                        : todo
                     )
                );
            };

            return (
                <div>
                    <h3>Todo List</h3>
                    
                    <input type="text" value={inputValue} onChange={handleChange} onKeyDown={(e) => {if (e.key === 'Enter') {handleAdd()}}}/>
                    <button onClick={handleAdd}>추가</button>

                    <div>
                        {todos.map(todo => (
                            <TodoItem
                                key={todo.id}
                                todo={todo}
                                onRemove={() => handleRemove(todo.id)}
                                onToggle={() => handleToggle(todo.id)}/>
                        ))}
                    </div>
                </div>
            );
        }

        function App() {
            return (
                <div>
                    <h1>오늘의 할일</h1>
                    <TodoList />
                </div>
            )
        }


        ReactDOM.createRoot(document.getElementById('root')).render(<App />);
    
    </script>
    
</body>
</html>
```

## 3. hooks

💡 Hooks는 리액트 16.8에서 새로 도입된 기능으로, **함수 컴포넌트**에서 사용 불가능한 생명주기 메소드의 한계점으로 인해 **상태 관리 및 렌더링 이후 시점 컨트롤 등 다양한 문제를 해결하기 위해 만든 함수 집합**을 의미한다.


</br>


### 3-1. useEffect
#### 3-1-1. Basic
</br>

* 함수형 컴포넌트에서는 생명주기 API 사용이 불가능하다.
* **useEffect**: 함수형 컴포넌트에서도 **렌더링 된 이후 시점에 수행 할 내용이 필요한 경우 사용할 수 있는 기능**을 hooks로 제공


</br>

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Document</title>
    <script crossorigin src="https://unpkg.com/react@18/umd/react.development.js"></script>
    <script crossorigin src="https://unpkg.com/react-dom@18/umd/react-dom.development.js"></script>
    <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>
</head>
<body>
    <div id="root"></div>

    <script type="text/babel">

        /* 함수형 컴포넌트에서 렌더링 된 이후 시점에 수행 할 내용이 필요한 경우
        사용할 수 있는 기능을 hooks로 제공해주고 있고 useEffect이다. */

        const { useEffect } = React;

        function MessagePrinter() {
            console.log('렌더링');

            useEffect(() => {
                console.log('렌더링 이후 동작...');
            })

            return (
                <h1>{console.log('렌더링 시 출력')}</h1>
            );

            // return문 아래에 있어 동작하지 않음
            // console.log('렌더링 이후 동작...');
        }


        ReactDOM.createRoot(document.getElementById('root')).render(<MessagePrinter />);
    
    </script>
    
</body>
</html>
```

#### 3-1-2. Mount

</br>

💡 마운트(Mount) : DOM이 생성되고 웹 브라우저상 나타나는 것

</br>

* mount 시 생명주기 메소드 라이프 사이클
(constructor) → (getDerivedStateFromProps) → (render) → (componentDidMount)


</br>

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Document</title>
    <script crossorigin src="https://unpkg.com/react@18/umd/react.development.js"></script>
    <script crossorigin src="https://unpkg.com/react-dom@18/umd/react-dom.development.js"></script>
    <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>
</head>
<body>
    <div id="root"></div>

    <script type="text/babel">

        const { useState, useEffect } = React;

        function TimePrinter() {

            const [time, setTime] = useState(new Date().toLocaleTimeString());

            console.log('렌더링...')

            useEffect(
                () => { console.log('useEffect 동작')}, // 수행할 동작 - 콜백 함수
                []  // 의존성 배열을 빈 배열로 설정하면 업데이트 시점에는 동작하지 않는다.
            )

            return(
                <>
                    <button onClick={() => setTime(new Date().toLocaleTimeString())}>현재 시간 확인하기</button>
                    <h1>{time}</h1>
                </>
            )
        }


        ReactDOM.createRoot(document.getElementById('root')).render(<TimePrinter />);
    
    </script>
    
</body>
</html>
```

#### 3-1-3. Lifecycle

</br>

💡 **라이프 사이클 메소드** : 컴포넌트를 통해 엘리먼트가 브라우저상에 나타나고, 업데이트되고, 사라지게 될 때 중간 중간 호출 되는 메소드들
*  총 9가지가 있다.
* *클래스형 컴포넌트에서만 사용이 가능하다.
* 상황 별로 라이프 사이클은 크게 총3가지 카테고리인 마운트, 업데이트, 언마운트로 나뉜다.
* 생명주기 도표 : https://projects.wojtekmaj.pl/react-lifecycle-methods-diagram/


</br></br>

💡 **Update**: 컴포넌트의 상태가 바뀌는 경우를 의미한다.
*  총 4가지의 업데이트 상황이 있다.
1. props 변경
2. state 변경
3. 부모 컴포넌트 리렌더링
4. this.forceUpdate로 강제 렌더링 트리거

</br>

* update 시 생명주기 메소드 라이프 사이클
(getDerivedStateFromProps) → (shouldComponentUpdate) → (render) → (getSnapshotBeforeUpdwate) → (componentDidUpdate)



</br></br>


💡 **언마운트(unmount)**: 컴포넌트를 DOM에서 제거하는 것
* 언마운트에서의 생명주기 메소드는 `componentWillUnmount`이며 언마운트 시 호출된다.



</br>

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Document</title>
    <script crossorigin src="https://unpkg.com/react@18/umd/react.development.js"></script>
    <script crossorigin src="https://unpkg.com/react-dom@18/umd/react-dom.development.js"></script>
    <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>
</head>
<body>
    <div id="root"></div>

    <script type="text/babel">

        const { useState, useEffect } = React;

        function Counter() {

            const [count, setCount] = useState(0);
            const [otherCount, setOtherCount] = useState(0);

            // useEffect는 첫 마운트 직후에는 무조건 한 번 실행 된다.

            // 1. 의존성 배열 생략 : 매 렌더링마다 실행
            useEffect(() => {
                // 렌더링 '완료된 후' 실행된다.
                console.log('렌더링될 때마다 실행')
                document.title = `넌 ${count}번 클릭했어`;
            })

            // 2. 빈 의존성 배열 [] : Mount(첫 렌더링) 시에만 실행
            // 주로 초기 데이터 로딩(API 호출) 등에 사용
            useEffect(() => {
                console.log('컴포넌트가 처음 화면에 나타남(Mount)');
            }, []);

            //3. 의존성 배열에 특정 값 지정 [count] : 해당 값이 변경될 때만 실행
            useEffect(() => {
                console.log(`count 값이 ${count}로 변경되어, 이 useEffect가 실행`)
            }, [count])


            console.log('Counter 컴포넌트 렌더링');


            return (
                <div>
                    <h2>단순 카운터</h2>
                    <p>{count}번 클릭 했습니다.</p>
                    <button onClick={() => setCount(count + 1)}>클릭</button>
                    <p>Other Count: {otherCount}</p>
                    <button onClick={ () => setOtherCount(otherCount +1)}>클릭</button>
                </div>
                
            )
        }

        function Timer() {

            useEffect(() => {
                console.log('타이머가 시작되었습니다(Mount)');

                const timerId = setInterval(() => {
                    console.log('...1초 경과...');
                }, 1000);

                // Cleanup 함수: useEffect가 반환하는 함수이다.
                // 컴포넌트가 화면에서 사라지기 직전(Unmount)에 호출된다.
                return () => {
                    console.log('타이머를 정리합니다(Unmount)')
                    clearInterval(timerId); // 생성했던 타이머 제거
                }
            }, []);

            return(
                <h3>타이머가 동작 중입니다. (콘솔 확인)</h3>
            )
        }

        function App() {
            const [showTimer, setShowTimer] = useState(true);

            return (
                <>
                    <Counter />
                    <hr />
                    <button onClick={() => setShowTimer(!showTimer)}>타이머 보이기/숨기기</button>
                    {showTimer && <Timer />}
                </>
            )
        }

        ReactDOM.createRoot(document.getElementById('root')).render(<App />);
    
    </script>
    
</body>
</html>
```

* map과 filter 메소드는 활용도가 높아서 정확히 알아야 할 것 같다. 그런데 어렵다..
