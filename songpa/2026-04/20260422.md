```html
<!DOCTYPE html>
<html lang="en">

</head>

    <script type="text/babel">

        const { useEffect, useState } = React;

        const API_URL = 'https://pokeapi.co/api/v2/pokemon';

        // 앞모습
        // sprites.other.dream_world.front_default

        function Title() {
            return <h1>포켓몬 목록</h1>
        }

        function PokemonList() {
            const [pokemons, setPokemons] = useState([]);     // 서버에서 받아올 포켓몬 목록
            const [loading, setLoading] = useState(true);   // 로딩 상태
            const [error, setError] = useState(null);           // 에러 상태

            useEffect(() => {
                 const fetchAllPokemonData = async () => {

                    // 1. 최초 포켓몬 목록(이름과 상세정보 URL)을 가져온다.
                    try{
                        const listResponse = await fetch(API_URL);

                        if(!listResponse.ok) throw new Error('포켓몬 목록을 가져올 수 없습니다.');

                        const listData = await listResponse.json();
                        // 2. 각 포켓몬의 상세 정보 fetch Promise 배열을 만든다.
                        // (map을 활용하여 상세 정보 url로 fetch 요청해서 return res.json(); )
                        const detailPromise = listData.results.map(pokemon => 
                            fetch(pokemon.url).then(res => {
                                if(!res.ok) throw new Error('정보를 가져올 수 없습니다.');
                                return res.json();
                            })
                        );

                        // 3. Promise.all로 모든 상세 정보 요청이 완료될 때까지 기다린다.
                        // 배열 안의 모든 Promise가 이행될 때까지 기다렸다가, 결과를 배열로 한꺼번에 반환한다.
                        const detailedPokemons = await Promise.all(detailPromise);

                        /* 참고: Promise.all()은 배열 안 Promise 중 하나라도 reject 되면 전체가 reject 됨
                        만약 일부가 실패해도 나머지는 계속 처리하고 싶다면 Promise.addSettled() 사용*/

                        // 4. 상태를 업데이트 한다.
                        setPokemons(detailedPokemons);

                    } catch(error) {
                        console.error(error);
                        setError(error);
                    } finally {
                        setLoading(false);
                    }
                 };
                 fetchAllPokemonData();
            }, []);

            if(loading) return <p>모든 포켓몬을 불러오는 중...</p>
            if (error) return <p>오류가 발생했습니다.: {error.message}</p>

            return (
                <div>
                    {pokemons.map(pokemon => <Card key={pokemon.id} pokemon={pokemon}/>)}    
                </div>
            )
        }

        function Card({pokemon}) {
            return (
                <div className="card">
                    <img src={pokemon.sprites.other.dream_world.front_default} alt={pokemon.name}/> 
                    <div className="info">
                        <h2>No. {pokemon.id}</h2>
                        <h3>name: {pokemon.name}</h3>
                    </div>
                </div>
            )
        }


        function App () {
            return (
                <>
                    <Title />
                    <PokemonList />
                </>
            )
        }

        ReactDOM.createRoot(document.getElementById('root')).render(<App />);
    
    </script>
    
</body>
</html>
```

#### 4-3-3. custom hook

</br>

* **'4-3-1 ItemList'** 의 코드를 custom hook를 사용하여 리팩토링해봄

</br>

* **AbortController**: 비동기 요청을 중간에 취소하기 위한 도구.

* **페이지 이동, 중복 요청 방지, 응답 시간이 너무 길 때** 등 통신이 중단될 경우 사용

</br>

* **작동 방식**: 컨트롤러 생성 -> 그 안의 signal을 요청 함수에 넘겨줌 -> 필요할 때 abort() 호출

</br>

* 리액트 개발자라면 **useEffect의 clean-up 함수**(컴포넌트가 사라질 때 실행되는 부분)에서 이 abort()를 호출해주는 것이 아주 좋은 습관암
* "내가 사라질 때 아직 안 끝난 통신이 있다면 다 끊어줘!"라고 예약하는 셈

</br>

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Document</title>
    <script crossorigin src="https://unpkg.com/react@18/umd/react.development.js"></script>
    <script crossorigin src="https://unpkg.com/react-dom@18/umd/react-dom.development.js"></script>
    <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>
    <style>
        .item {
            background-color: bisque;
            border: 1px dotted black;
            width: 350px;
            display: inline-block;
            margin: 10px;
        }
    </style>
</head>
<body>
    <div id="root"></div>

    <script type="text/babel">

        const { useEffect, useState } = React;

        function useFetch(url) {
            const [data, setData] = useState(null);     // 서버에서 받아올 목록
            const [loading, setLoading] = useState(true);   // 로딩 상태
            const [error, setError] = useState(null);       //에러 상태

            useEffect(() => {

                // 통신 강제 취소 스위치 생성
                const controller = new AbortController();

                // fetch 옵션에 취소 스위치를 연결
                fetch(url, {signal: controller.signal})
                    .then(response => {
                        if (!response.ok) {
                            throw new Error('네트워크 응답 문제가 있습니다.')
                        }
                        return response.json();
                    })
                    .then(responseData => setData(responseData))
                    .catch(error => {
                        if(error.name === 'AbortError') {
                            console.log('통신이 정상적으로 중단되었습니다.');
                        } else {
                            console.error('데이터를 불러오는 데 실패했습니다.', error);
                            setError(error);    //에러 발생 시 에러 상태에 저장
                        }
                    })
                    .finally(() => setLoading(false));

                    // 클린업 함수: 이 훅을 쓰던 컴포넌트가 화면에서 사라지면 즉시 통신 취소
                    return () => controller.abort();

            }, [url]);

            return { data, loading, error }
        }

        function Title() {
            return <h1>회원 목록</h1>;
        }

        function ItemList() {

            const {data: users, loading, error} = useFetch('https://jsonplaceholder.typicode.com/users')


            // Early Return 패턴
            if (loading) {
                return <p>로딩 중...</p>;
            }
            if (error) {
                return <p>오류가 발생했습니다: {error.message} </p>;
            }
            if (!users) {
                return <h3>데이터가 없습니다.</h3>;
            }

            return (
                <div>
                    {users.map(user => <Item key={user.id} user={user}/>)}
                </div>
            )

        }

        function Item({user}) {
            return (
                <div className="item">
                    <h4>{user.name}</h4>
                    user Id : {user.id} <br/>
                    email : {user.email} <br/>
                    phone : {user.phone} <br/>
                    {/* 옵셔널 체이닝 연산자 사용(에러 방지) */}
                    company name: {user.company?.name} 
                </div>
            );
        }

        function App () {
            return (
                <>
                    <Title />
                    <ItemList />
                </>
            )
        }

        ReactDOM.createRoot(document.getElementById('root')).render(<App />);
    
    </script>
    
</body>
</html>
```

### 5-3. Nexts param

</br>

**<주요 개념>**

💡**Next에 나오는 추가적인 훅 (주요 4가지)**

※ 클라이언트 컴포넌트에서만 사용 가능!  파일 상단에 `'use client'` 꼭 붙여주기

</br>

* `usePathname()` : 현재 URL 경로를 읽어옴 -> 활성 탭/ 메뉴 스타일 적용 시 사용
* `useParams()` : [폴더명] 형태의 동적 라우팅 파라미터를 읽어온다 -> 상세페이지 (/menu/1) 데이터 패칭 시 사용
* `useSearchParams()` : URL 끝의 쿼리 스트링 (?key=value)를 읽어옴 -> 검색 결과 페이지, 필터링
* `useRouter()` : 함수/이벤트 내에서 프로그래밍 방식으로 페이지 이동 -> 폼 제출 완료페이지 이동, 뒤로가기 등
